Quick Start
# 1. Start Python NLP (port 8000)
cd python-microservice
pip install -r requirements.txt
uvicorn main:app --host 0.0.0.0 --port 8000

# 2. Start Requirement Analyzer (port 8081)
cd Requirement-Analyzer
mvn spring-boot:run

# 3. Start Code Analyzer (port 8082)
cd Code-Analyzer/backend
mvn spring-boot:run

# 4. Start API Gateway (port 8085)
cd api-gateway
mvn spring-boot:run

# 5. Start Sonar Analyzer (port 8083)
cd Backend/Sonar-Analyzer/backend
node server.js

# All services are now available via http://localhost:8085/api/*
# Sonar Analyzer available at http://localhost:8083
